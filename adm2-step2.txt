@Injectable()

	// ---------------------------------------------------------------- roles
	async listRoles(): Promise<Role[]> {
		const roleRows = await db.select().from(roles).orderBy(roles.name)

		const permRows = await db
			.select({ roleId: rolePermissions.roleId, name: permissions.name })
			.from(rolePermissions)
			.innerJoin(permissions, eq(rolePermissions.permissionId, permissions.id))

		const countRows = await db
			.select({ roleId: userRoles.roleId, count: sql<number>`count(*)::int` })
			.from(userRoles)
			.groupBy(userRoles.roleId)

		const permsByRole = new Map<number, PermissionName[]>()
		for (const row of permRows) {
			if (!PERMISSION_CATALOG.has(row.name)) continue
			const list = permsByRole.get(row.roleId) ?? []
			list.push(row.name as PermissionName)
			permsByRole.set(row.roleId, list)
		}

		const countByRole = new Map<number, number>()
		for (const row of countRows) {
			countByRole.set(row.roleId, row.count)
		}

		return roleRows.map(role => ({
			id: role.id,
			name: role.name,
			description: role.description,
			permissions: this.orderPermissions(permsByRole.get(role.id) ?? []),
			userCount: countByRole.get(role.id) ?? 0,
			createdAt: role.createdAt,
			updatedAt: role.updatedAt,
		}))
	}

	async createRole(payload: CreateRoleInput, actorId: string | null): Promise<Role> {
		const roleId = await db.transaction(async tx => {
			const [existing] = await tx.select().from(roles).where(eq(roles.name, payload.name))
			if (existing) {
				throw new ORPCError("CONFLICT", { message: `Role "${payload.name}" already exists` })
			}

			const [role] = await tx
				.insert(roles)
				.values({ name: payload.name, description: payload.description ?? null })
				.returning()

			if (!role) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", { message: "Role could not be created" })
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_RBAC,
					action: "role.create",
					actorId,
					targetType: "role",
					targetId: String(role.id),
					oldValue: null,
					newValue: { name: role.name, description: role.description },
				},
				tx
			)

			return role.id
		})

		return this.buildRole(roleId)
	}

	async updateRole(payload: UpdateRoleRequest, actorId: string | null): Promise<Role> {
		const role = await this.getRoleRowOrThrow(payload.id)

		if (role.name === ADMIN_ROLE && payload.name !== undefined && payload.name !== ADMIN_ROLE) {
			await this.auditService.record({
				domain: AUDIT_DOMAIN_RBAC,
				action: "role.update",
				outcome: "denied",
				actorId,
				targetType: "role",
				targetId: String(role.id),
				oldValue: { name: role.name },
				newValue: { name: payload.name },
				reason: "The Admin role cannot be renamed",
			})
			throw new ORPCError("FORBIDDEN", { message: "The Admin role cannot be renamed" })
		}

		await db.transaction(async tx => {
			if (payload.name !== undefined && payload.name !== role.name) {
				const [clash] = await tx.select().from(roles).where(eq(roles.name, payload.name))
				if (clash) {
					throw new ORPCError("CONFLICT", { message: `Role "${payload.name}" already exists` })
				}
			}

			const next = {
				name: payload.name ?? role.name,
				description: payload.description !== undefined ? payload.description : role.description,
			}

			await tx
				.update(roles)
				.set({ ...next, updatedAt: new Date() })
				.where(eq(roles.id, role.id))

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_RBAC,
					action: "role.update",
					actorId,
					targetType: "role",
					targetId: String(role.id),
					oldValue: { name: role.name, description: role.description },
					newValue: next,
				},
				tx
			)
		})

		return this.buildRole(role.id)
	}

	async setRolePermissions(
		payload: SetRolePermissionsRequest,
		actorId: string | null
	): Promise<Role> {
		const role = await this.getRoleRowOrThrow(payload.id)

		const uniqueNames = Array.from(new Set(payload.permissions))
		const permissionRows = uniqueNames.length
			? await db.select().from(permissions).where(inArray(permissions.name, uniqueNames))
			: []

		const foundNames = new Set(permissionRows.map(p => p.name))
		const missing = uniqueNames.filter(name => !foundNames.has(name))
		if (missing.length) {
			throw new ORPCError("BAD_REQUEST", { message: `Unknown permissions: ${missing.join(", ")}` })
		}

		const previous = await db
			.select({ name: permissions.name })
			.from(rolePermissions)
			.innerJoin(permissions, eq(rolePermissions.permissionId, permissions.id))
			.where(eq(rolePermissions.roleId, role.id))

		await db.transaction(async tx => {
			await tx.delete(rolePermissions).where(eq(rolePermissions.roleId, role.id))
			if (permissionRows.length) {
				await tx
					.insert(rolePermissions)
					.values(permissionRows.map(p => ({ roleId: role.id, permissionId: p.id })))
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_RBAC,
					action: "role.setPermissions",
					actorId,
					targetType: "role",
					targetId: String(role.id),
					oldValue: { permissions: previous.map(p => p.name).sort() },
					newValue: { permissions: uniqueNames.slice().sort() },
				},
				tx
			)
		})

		await this.cacheService.clear()
		return this.buildRole(role.id)
	}

	async deleteRole(id: number, actorId: string | null): Promise<{ success: boolean; id: number }> {
		const role = await this.getRoleRowOrThrow(id)

		if (role.name === ADMIN_ROLE) {
			await this.auditService.record({
				domain: AUDIT_DOMAIN_RBAC,
				action: "role.delete",
				outcome: "denied",
				actorId,
				targetType: "role",
				targetId: String(role.id),
				oldValue: { name: role.name },
				reason: "The Admin role cannot be deleted",
			})
			throw new ORPCError("FORBIDDEN", { message: "The Admin role cannot be deleted" })
		}

		const grantedPermissions = await db
			.select({ name: permissions.name })
			.from(rolePermissions)
			.innerJoin(permissions, eq(rolePermissions.permissionId, permissions.id))
			.where(eq(rolePermissions.roleId, role.id))

		await db.transaction(async tx => {
			await tx.delete(roles).where(eq(roles.id, role.id))

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_RBAC,
					action: "role.delete",
					actorId,
					targetType: "role",
					targetId: String(role.id),
					oldValue: {
						name: role.name,
						description: role.description,
						permissions: grantedPermissions.map(p => p.name).sort(),
					},
					newValue: null,
				},
				tx
			)
		})

		await this.cacheService.clear()
		return { success: true, id: role.id }
	}

	// ---------------------------------------------------------- permissions
	async listPermissions(): Promise<PermissionCatalogEntry[]> {
		const rows = await db.select().from(permissions)
		const descByName = new Map(rows.map(row => [row.name, row.description]))

		return PERMISSION_NAMES.filter(name => descByName.has(name)).map(name => ({
			name,
			description: descByName.get(name) ?? null,
		}))
	}

	// ---------------------------------------------------------------- users
	async listUsers(query: ListUsersRequest, includeDirectory: boolean): Promise<ListUsersResponse> {
		const { items } = await this.queryFilteredUsers(query, includeDirectory)
		const total = items.length
		const pageSize = query.pageSize ?? 20
		const totalPages = Math.max(1, Math.ceil(total / pageSize))
		const page = Math.min(query.page ?? 1, totalPages)
		const offset = (page - 1) * pageSize

		return {
			items: items.slice(offset, offset + pageSize),
			total,
			page,
			pageSize,
			totalPages,
		}
	}

	/**
	 * Export the filtered user list as CSV, XLSX or PDF.
	 *
	 * A bulk export of personal data, so every export is audited (who, which
	 * format and filters, how many rows) — never the names or emails exported,
	 * and never the search text, which is often a person's name.
	 */
	async exportUsers(
		query: ExportUsersRequest,
		includeDirectory: boolean,
		actorId: string | null
	): Promise<ExportUsersResponse> {
		const { items } = await this.queryFilteredUsers(query, includeDirectory)
		const rows = items.map(user => ({
			name: user.name ?? "",
			email: includeDirectory ? user.email : "[redacted]",
			accountType: user.accountType,
			status: user.status,
			roles: user.roles.join(", "),
			createdAt: user.createdAt.toISOString(),
		}))

		const dateStamp = new Date().toISOString().slice(0, 10)
		const exported = await this.exportService.exportTable({
			format: query.format,
			fileBaseName: `qsps-users-${dateStamp}`,
			title: "QSPS User Accounts",
			columns: [
				{ key: "name", header: "Name" },
				{ key: "email", header: "Email" },
				{ key: "accountType", header: "Account Type" },
				{ key: "status", header: "Status" },
				{ key: "roles", header: "Roles" },
				{ key: "createdAt", header: "Created At (UTC)" },
			],
			rows,
		})

		// Nothing to roll back: the export only reads, so the entry stands alone.
		await this.auditService.record({
			domain: AUDIT_DOMAIN_RBAC,
			action: "user.export",
			actorId,
			targetType: "user",
			targetId: null,
			newValue: {
				format: query.format,
				filters: {
					status: query.status ?? null,
					accountType: query.accountType ?? null,
					role: query.role ?? null,
					search: Boolean(query.search),
				},
				includesEmails: includeDirectory,
				rowCount: rows.length,
			},
		})

		return {
			fileName: exported.fileName,
			mimeType: exported.mimeType,
			contentBase64: exported.buffer.toString("base64"),
		}
	}

	async createUser(payload: CreateStaffUserInput, actorId: string | null): Promise<UserWithRoles> {
		const normalizedEmail = payload.email.trim().toLowerCase()
		const uniqueRoleNames = Array.from(new Set(payload.roles))

		const createdUser = await this.runUserGuarded(
			{
				action: "user.create",
				actorId,
				targetId: normalizedEmail,
				attempted: { accountType: "staff", roles: uniqueRoleNames },
			},
			async tx => {
				const roleRows = await tx.select().from(roles).where(inArray(roles.name, uniqueRoleNames))

				const foundRoleNames = new Set(roleRows.map(r => r.name))
				const missingRoles = uniqueRoleNames.filter(name => !foundRoleNames.has(name))
				if (missingRoles.length > 0) {
					throw new ORPCError("BAD_REQUEST", {
						message: `Unknown roles: ${missingRoles.join(", ")}`,
					})
				}

				if (uniqueRoleNames.includes(ADMIN_ROLE)) {
					const actorIsAdmin = actorId ? await this.rbacService.holdsAdmin(tx, actorId) : false
					if (!actorIsAdmin) {
						throw new PrivilegeGuardError(
							"FORBIDDEN",
							"Only a Super Admin may assign the Super Admin role",
							"caller is not a Super Admin"
						)
					}
				}

				const [existing] = await tx
					.select({ id: users.id })
					.from(users)
					.where(eq(users.email, normalizedEmail))
				if (existing) {
					throw new ORPCError("CONFLICT", {
						message: "An account with this email address already exists",
					})
				}

				const userId = randomUUID()
				const now = new Date()
				const [inserted] = await tx
					.insert(users)
					.values({
						id: userId,
						name: payload.name.trim(),
						email: normalizedEmail,
						emailVerified: true,
						accountType: "staff",
						mustResetPassword: true,
						createdAt: now,
						updatedAt: now,
					})
					.returning()

				if (!inserted) {
					throw new ORPCError("INTERNAL_SERVER_ERROR", {
						message: "Staff account could not be created",
					})
				}

				await tx.insert(userRoles).values(
					roleRows.map(role => ({
						userId: inserted.id,
						roleId: role.id,
					}))
				)

				// Rule 5: Never record personal data (name, email, mobile, birthDate) in audit values.
				await this.auditService.record(
					{
						domain: AUDIT_DOMAIN_RBAC,
						action: "user.create",
						actorId,
						targetType: "user",
						targetId: inserted.id,
						oldValue: null,
						newValue: {
							accountType: "staff",
							mustResetPassword: true,
							roles: uniqueRoleNames.slice().sort(),
						},
					},
					tx
				)

				return {
					id: inserted.id,
					name: inserted.name,
					email: inserted.email,
					accountType: "staff" as const,
					status: "active" as const,
					roles: uniqueRoleNames.slice().sort(),
					disabledAt: null,
					deletedAt: null,
					createdAt: inserted.createdAt,
				}
			}
		)

		await this.sendStaffInvitationEmail(createdUser.email)
		return createdUser
	}

	async updateUser(payload: UpdateUserRequest, actorId: string | null): Promise<UserWithRoles> {
		const updated = await db.transaction(async tx => {
			const [current] = await tx
				.select()
				.from(users)
				.where(and(eq(users.id, payload.userId), isNull(users.deletedAt)))
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `User "${payload.userId}" was not found` })
			}

			const changedFields: string[] = []
			const updates: Partial<typeof users.$inferInsert> = { updatedAt: new Date() }

			if (payload.name !== undefined && payload.name.trim() !== current.name) {
				updates.name = payload.name.trim()
				changedFields.push("name")
			}

			if (payload.email !== undefined) {
				const nextEmail = payload.email.trim().toLowerCase()
				if (nextEmail !== current.email.toLowerCase()) {
					const [clash] = await tx
						.select({ id: users.id })
						.from(users)
						.where(eq(users.email, nextEmail))
					if (clash) {
						throw new ORPCError("CONFLICT", {
							message: "An account with this email address already exists",
						})
					}
					updates.email = nextEmail
					changedFields.push("email")
				}
			}

			const [saved] =
				changedFields.length > 0
					? await tx.update(users).set(updates).where(eq(users.id, current.id)).returning()
					: [current]

			if (!saved) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", { message: "User could not be updated" })
			}

			// Rule 5: Record which fields changed, never personal data values.
			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_RBAC,
					action: "user.update",
					actorId,
					targetType: "user",
					targetId: current.id,
					oldValue: { fields: changedFields },
					newValue: { changedFields },
				},
				tx
			)

			const roleRows = await tx
				.select({ name: roles.name })
				.from(userRoles)
				.innerJoin(roles, eq(userRoles.roleId, roles.id))
				.where(eq(userRoles.userId, saved.id))

			return this.toUserWithRoles(
				saved,
				roleRows.map(r => r.name),
				true
			)
		})

		return updated
	}

	async setUserStatus(
		payload: SetUserStatusRequest,
		actorId: string | null
	): Promise<UserWithRoles> {
		const action = payload.isActive ? "user.enable" : "user.disable"

		const updated = await this.runUserGuarded(
			{
				action,
				actorId,
				targetId: payload.userId,
				attempted: { isActive: payload.isActive },
			},
			async tx => {
				const [current] = await tx
					.select()
					.from(users)
					.where(and(eq(users.id, payload.userId), isNull(users.deletedAt)))
				if (!current) {
					throw new ORPCError("NOT_FOUND", { message: `User "${payload.userId}" was not found` })
				}

				if (!payload.isActive) {
					await this.rbacService.assertCanDeactivateUser(tx, current.id, actorId)
				}

				const now = new Date()
				const nextDisabledAt = payload.isActive ? null : (current.disabledAt ?? now)

				const [saved] = await tx
					.update(users)
					.set({ disabledAt: nextDisabledAt, updatedAt: now })
					.where(eq(users.id, current.id))
					.returning()

				if (!saved) {
					throw new ORPCError("INTERNAL_SERVER_ERROR", {
						message: "User status could not be updated",
					})
				}

				// Revoke all active sessions immediately upon disabling.
				if (!payload.isActive) {
					await tx.delete(sessions).where(eq(sessions.userId, current.id))
				}

				await this.auditService.record(
					{
						domain: AUDIT_DOMAIN_RBAC,
						action,
						actorId,
						targetType: "user",
						targetId: current.id,
						oldValue: { status: current.disabledAt ? "disabled" : "active" },
						newValue: { status: payload.isActive ? "active" : "disabled" },
					},
					tx
				)

				const roleRows = await tx
					.select({ name: roles.name })
					.from(userRoles)
					.innerJoin(roles, eq(userRoles.roleId, roles.id))
					.where(eq(userRoles.userId, saved.id))

				return this.toUserWithRoles(
					saved,
					roleRows.map(r => r.name),
					true
				)
			}
		)

		await this.cacheService.invalidate(payload.userId)
		return updated
	}

	async deleteUser(userId: string, actorId: string | null): Promise<DeleteUserResponse> {
		await this.runUserGuarded(
			{
				action: "user.delete",
				actorId,
				targetId: userId,
				attempted: { deleted: true },
			},
			async tx => {
				const [current] = await tx
					.select()
					.from(users)
					.where(and(eq(users.id, userId), isNull(users.deletedAt)))
				if (!current) {
					throw new ORPCError("NOT_FOUND", { message: `User "${userId}" was not found` })
				}

				await this.rbacService.assertCanDeactivateUser(tx, current.id, actorId)

				const now = new Date()
				// Soft delete: retain the user row, history, and audit entries.
				await tx
					.update(users)
					.set({
						deletedAt: now,
						disabledAt: current.disabledAt ?? now,
						updatedAt: now,
					})
					.where(eq(users.id, current.id))

				// Revoke all active sessions so the deleted user cannot stay signed in.
				await tx.delete(sessions).where(eq(sessions.userId, current.id))

				await this.auditService.record(
					{
						domain: AUDIT_DOMAIN_RBAC,
						action: "user.delete",
						actorId,
						targetType: "user",
						targetId: current.id,
						oldValue: { status: current.disabledAt ? "disabled" : "active" },
						newValue: { status: "deleted" },
					},
					tx
				)
			}
		)

		await this.cacheService.invalidate(userId)
		return { success: true, userId }
	}

	async assignUserRole(payload: AssignRoleRequest, actorId: string | null) {

	async removeUserRole(payload: RemoveRoleRequest, actorId: string | null) {

		return { success: true, userId: payload.userId, roleName: payload.roleName }
	}

	// -------------------------------------------------------------- helpers
	private async queryFilteredUsers(
		filters: UserFilterInput,
		includeDirectory: boolean
	): Promise<{ items: UserWithRoles[] }> {
		const conditions: SQL[] = []

		// Exclude soft-deleted users from the default list unless explicitly filtered.
		if (filters.status === "deleted") {
			conditions.push(isNotNull(users.deletedAt))
		} else if (filters.status === "disabled") {
			conditions.push(isNull(users.deletedAt), isNotNull(users.disabledAt))
		} else if (filters.status === "active") {
			conditions.push(isNull(users.deletedAt), isNull(users.disabledAt))
		} else {
			conditions.push(isNull(users.deletedAt))
		}

		if (filters.accountType) {
			conditions.push(eq(users.accountType, filters.accountType))
		}

		if (filters.search) {
			const pattern = `%${filters.search}%`
			const searchCondition = includeDirectory
				? or(ilike(users.name, pattern), ilike(users.email, pattern))
				: ilike(users.name, pattern)
			if (searchCondition) {
				conditions.push(searchCondition)
			}
		}

		const userRows = await db
			.select()
			.from(users)
			.where(and(...conditions))
			.orderBy(users.name, users.email)

		if (userRows.length === 0) {
			return { items: [] }
		}

		const userIds = userRows.map(u => u.id)
		const roleRows = await db
			.select({ userId: userRoles.userId, name: roles.name })
			.from(userRoles)
			.innerJoin(roles, eq(userRoles.roleId, roles.id))
			.where(inArray(userRoles.userId, userIds))

		const rolesByUser = new Map<string, string[]>()
		for (const row of roleRows) {
			const list = rolesByUser.get(row.userId) ?? []
			list.push(row.name)
			rolesByUser.set(row.userId, list)
		}

		let items = userRows.map(user =>
			this.toUserWithRoles(user, rolesByUser.get(user.id) ?? [], includeDirectory)
		)

		if (filters.role) {
			items = items.filter(user => user.roles.includes(filters.role!))
		}

		return { items }
	}

	private toUserWithRoles(
		user: typeof users.$inferSelect,
		assignedRoles: string[],
		includeDirectory: boolean
	): UserWithRoles {
		const status: UserAccountStatus = user.deletedAt
			? "deleted"
			: user.disabledAt
				? "disabled"
				: "active"
		const accountType: UserAccountType = user.accountType === "staff" ? "staff" : "client"

		return {
			id: user.id,
			name: user.name,
			email: includeDirectory ? user.email : "",
			accountType,
			status,
			roles: assignedRoles.slice().sort(),
			disabledAt: user.disabledAt,
			deletedAt: user.deletedAt,
			createdAt: user.createdAt,
		}
	}

	private async sendStaffInvitationEmail(email: string): Promise<void> {
		try {
			const { authEnv, getAuth } = await import("@repo/auth")
			const auth = getAuth()
			await auth.api.requestPasswordReset({
				body: {
					email,
					redirectTo: `${authEnv.APP_WEB_URL}/reset-password`,
				},
			})
		} catch {
			// Delivery errors are logged inside Better Auth's sendResetPassword handler;
			// the staff account is already committed and can be re-invited if needed.
		}
	}

	private async runUserGuarded<T>(
		context: {
			action: string
			actorId: string | null
			targetId: string
			attempted: Record<string, unknown>
		},
		work: (tx: Parameters<Parameters<typeof db.transaction>[0]>[0]) => Promise<T>
	): Promise<T> {
		try {
			return await db.transaction(work)
		} catch (error) {
			if (!(error instanceof PrivilegeGuardError)) throw error

			await this.auditService.record({
				domain: AUDIT_DOMAIN_RBAC,
				action: context.action,
				outcome: "denied",
				actorId: context.actorId,
				targetType: "user",
				targetId: context.targetId,
				newValue: context.attempted,
				reason: error.reason,
			})

			throw new ORPCError(error.kind, { message: error.message })
		}
	}

	private async getRoleRowOrThrow(id: number) {
		const [role] = await db.select().from(roles).where(eq(roles.id, id))
		if (!role) {
			throw new ORPCError("NOT_FOUND", { message: `Role with ID ${id} not found` })
		}
		return role
	}

	private async buildRole(id: number): Promise<Role> {
		const role = await this.getRoleRowOrThrow(id)

		const permRows = await db
			.select({ name: permissions.name })
			.from(rolePermissions)
			.innerJoin(permissions, eq(rolePermissions.permissionId, permissions.id))
			.where(eq(rolePermissions.roleId, id))

		const [{ count } = { count: 0 }] = await db
			.select({ count: sql<number>`count(*)::int` })
			.from(userRoles)
			.where(eq(userRoles.roleId, id))

		const grantedPermissions = permRows
			.map(p => p.name)
			.filter((name): name is PermissionName => PERMISSION_CATALOG.has(name))

		return {
			id: role.id,
			name: role.name,
			description: role.description,
			permissions: this.orderPermissions(grantedPermissions),
			userCount: count,
			createdAt: role.createdAt,
			updatedAt: role.updatedAt,
		}
	}

	private orderPermissions(names: PermissionName[]): PermissionName[] {
		const held = new Set(names)
		return PERMISSION_NAMES.filter(name => held.has(name))
	}
}

import { Injectable } from "@nestjs/common"
import { ORPCError } from "@orpc/server"
import { eq } from "drizzle-orm"

import type { CreateLocationInput, Location, UpdateLocationRequest } from "@repo/contracts"
import { locations } from "@repo/db/schema"

import { AuditService } from "@/common/audit/audit.service"
import { db } from "@/common/database/database.client"

const AUDIT_DOMAIN_LOCATIONS = "locations"

@Injectable()
export class LocationsService {
	constructor(private readonly auditService: AuditService) {}

	async list(): Promise<Location[]> {
		return db.select().from(locations).orderBy(locations.name)
	}

	async listAvailable(): Promise<Location[]> {
		return db.select().from(locations).where(eq(locations.isActive, true)).orderBy(locations.name)
	}

	async create(input: CreateLocationInput, actorId: string | null): Promise<Location> {
		return db.transaction(async tx => {
			const [location] = await tx
				.insert(locations)
				.values({
					...input,
					isActive: true,
				})
				.onConflictDoNothing({ target: locations.code })
				.returning()

			if (!location) {
				throw new ORPCError("CONFLICT", {
					message: `Location code "${input.code}" is already in use`,
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_LOCATIONS,
					action: "location.create",
					actorId,
					targetType: "location",
					targetId: location.code,
					oldValue: null,
					newValue: this.auditValue(location),
				},
				tx
			)

			return location
		})
	}

	async update(input: UpdateLocationRequest, actorId: string | null): Promise<Location> {
		return db.transaction(async tx => {
			const [current] = await tx.select().from(locations).where(eq(locations.code, input.code))
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Location "${input.code}" was not found` })
			}

			const next = {
				name: input.name ?? current.name,
				address: input.address ?? current.address,
				region: input.region ?? current.region,
			}

			const [updated] = await tx
				.update(locations)
				.set({ ...next, updatedAt: new Date() })
				.where(eq(locations.code, current.code))
				.returning()
			if (!updated) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Location could not be updated",
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_LOCATIONS,
					action: "location.update",
					actorId,
					targetType: "location",
					targetId: current.code,
					oldValue: this.auditValue(current),
					newValue: this.auditValue(updated),
				},
				tx
			)

			return updated
		})
	}

	async setActive(
		input: { code: string; isActive: boolean },
		actorId: string | null
	): Promise<Location> {
		return db.transaction(async tx => {
			const [current] = await tx.select().from(locations).where(eq(locations.code, input.code))
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Location "${input.code}" was not found` })
			}

			const [updated] = await tx
				.update(locations)
				.set({ isActive: input.isActive, updatedAt: new Date() })
				.where(eq(locations.code, current.code))
				.returning()
			if (!updated) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Location status could not be updated",
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_LOCATIONS,
					action: input.isActive ? "location.enable" : "location.disable",
					actorId,
					targetType: "location",
					targetId: current.code,
					oldValue: this.auditValue(current),
					newValue: this.auditValue(updated),
				},
				tx
			)

			return updated
		})
	}

	async delete(code: string, actorId: string | null): Promise<never> {
		const [location] = await db.select().from(locations).where(eq(locations.code, code))
		if (!location) {
			throw new ORPCError("NOT_FOUND", { message: `Location "${code}" was not found` })
		}

		const reason =
			"Locations cannot be safely deleted because booking references cannot yet be checked. Disable the location instead."
		await this.auditService.record({
			domain: AUDIT_DOMAIN_LOCATIONS,
			action: "location.delete",
			outcome: "denied",
			actorId,
			targetType: "location",
			targetId: code,
			oldValue: this.auditValue(location),
			newValue: null,
			reason,
		})
		throw new ORPCError("CONFLICT", { message: reason })
	}

	private auditValue(location: Location) {
		return {
			code: location.code,
			name: location.name,
			address: location.address,
			region: location.region,
			isActive: location.isActive,
		}
	}
}

import { Injectable } from "@nestjs/common"
import { ORPCError } from "@orpc/server"
import { and, asc, eq, ne } from "drizzle-orm"

import type {
	CreateServiceInput,
	Service,
	ServiceCategory,
	UpdateServiceRequest,
} from "@repo/contracts"
import { serviceCategories, services } from "@repo/db/schema"

import { AuditService } from "@/common/audit/audit.service"
import { db } from "@/common/database/database.client"

const AUDIT_DOMAIN_SERVICES = "services"

type ServiceRow = typeof services.$inferSelect
type CategoryRow = typeof serviceCategories.$inferSelect

function serviceResponse(service: ServiceRow, category: CategoryRow): Service {
	return {
		id: service.id,
		serviceCategoryId: service.serviceCategoryId,
		categoryCode: category.code,
		categoryName: category.name,
		code: service.code,
		name: service.name,
		feeCentavos: service.feeCentavos,
		requiresApproval: service.requiresApproval,
		documentType: service.documentType,
		autoIssue: service.autoIssue,
		isActive: service.isActive,
		createdAt: service.createdAt,
		updatedAt: service.updatedAt,
	}
}

/**
 * Service catalog (ADM-5). Fees are integer centavos and apply to new Bookings
 * only; every change is audited with old and new values in the same transaction.
 */
@Injectable()
export class ServicesService {
	constructor(private readonly auditService: AuditService) {}

	async listCategories(): Promise<ServiceCategory[]> {
		return db.select().from(serviceCategories).orderBy(asc(serviceCategories.name))
	}

	async list(): Promise<Service[]> {
		const rows = await db
			.select({ service: services, category: serviceCategories })
			.from(services)
			.innerJoin(serviceCategories, eq(services.serviceCategoryId, serviceCategories.id))
			.orderBy(asc(serviceCategories.name), asc(services.name))
		return rows.map(({ service, category }) => serviceResponse(service, category))
	}

	async listAvailable(): Promise<Service[]> {
		const rows = await db
			.select({ service: services, category: serviceCategories })
			.from(services)
			.innerJoin(serviceCategories, eq(services.serviceCategoryId, serviceCategories.id))
			.where(eq(services.isActive, true))
			.orderBy(asc(serviceCategories.name), asc(services.name))
		return rows.map(({ service, category }) => serviceResponse(service, category))
	}

	async create(input: CreateServiceInput, actorId: string | null): Promise<Service> {
		return db.transaction(async tx => {
			const [category] = await tx
				.select()
				.from(serviceCategories)
				.where(eq(serviceCategories.id, input.serviceCategoryId))
			if (!category) {
				throw new ORPCError("NOT_FOUND", {
					message: `Service category "${input.serviceCategoryId}" was not found`,
				})
			}

			const [sameCode] = await tx
				.select({ id: services.id })
				.from(services)
				.where(eq(services.code, input.code))
				.limit(1)
			if (sameCode) {
				throw new ORPCError("CONFLICT", {
					message: `Service code "${input.code}" is already in use`,
				})
			}

			const [service] = await tx
				.insert(services)
				.values({
					serviceCategoryId: input.serviceCategoryId,
					code: input.code,
					name: input.name,
					feeCentavos: input.feeCentavos,
					requiresApproval: input.requiresApproval,
					documentType: input.documentType ?? null,
					autoIssue: input.autoIssue ?? true,
					isActive: true,
				})
				.onConflictDoNothing()
				.returning()
			if (!service) {
				throw new ORPCError("CONFLICT", {
					message: `A service named "${input.name}" already exists in this category`,
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SERVICES,
					action: "service.create",
					actorId,
					targetType: "service",
					targetId: String(service.id),
					oldValue: null,
					newValue: this.auditValue(service),
				},
				tx
			)
			return serviceResponse(service, category)
		})
	}

	async update(input: UpdateServiceRequest, actorId: string | null): Promise<Service> {
		return db.transaction(async tx => {
			const [current] = await tx.select().from(services).where(eq(services.id, input.id))
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Service "${input.id}" was not found` })
			}

			const serviceCategoryId = input.serviceCategoryId ?? current.serviceCategoryId
			const [category] = await tx
				.select()
				.from(serviceCategories)
				.where(eq(serviceCategories.id, serviceCategoryId))
			if (!category) {
				throw new ORPCError("NOT_FOUND", {
					message: `Service category "${serviceCategoryId}" was not found`,
				})
			}

			const name = input.name ?? current.name
			const [duplicate] = await tx
				.select({ id: services.id })
				.from(services)
				.where(
					and(
						eq(services.serviceCategoryId, serviceCategoryId),
						eq(services.name, name),
						ne(services.id, current.id)
					)
				)
				.limit(1)
			if (duplicate) {
				throw new ORPCError("CONFLICT", {
					message: `A service named "${name}" already exists in this category`,
				})
			}

			const [updated] = await tx
				.update(services)
				.set({
					serviceCategoryId,
					name,
					feeCentavos: input.feeCentavos ?? current.feeCentavos,
					requiresApproval: input.requiresApproval ?? current.requiresApproval,
					// undefined = leave as is; null = clear it.
					documentType:
						input.documentType === undefined ? current.documentType : input.documentType,
					autoIssue: input.autoIssue ?? current.autoIssue,
					updatedAt: new Date(),
				})
				.where(eq(services.id, current.id))
				.returning()
			if (!updated) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", { message: "Service could not be updated" })
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SERVICES,
					action:
						updated.feeCentavos !== current.feeCentavos ? "service.fee.update" : "service.update",
					actorId,
					targetType: "service",
					targetId: String(current.id),
					oldValue: this.auditValue(current),
					newValue: this.auditValue(updated),
				},
				tx
			)
			return serviceResponse(updated, category)
		})
	}

	async setActive(
		input: { id: number; isActive: boolean },
		actorId: string | null
	): Promise<Service> {
		return db.transaction(async tx => {
			const [current] = await tx.select().from(services).where(eq(services.id, input.id))
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Service "${input.id}" was not found` })
			}

			const [category] = await tx
				.select()
				.from(serviceCategories)
				.where(eq(serviceCategories.id, current.serviceCategoryId))
			if (!category) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Service category could not be loaded",
				})
			}

			const [updated] = await tx
				.update(services)
				.set({ isActive: input.isActive, updatedAt: new Date() })
				.where(eq(services.id, current.id))
				.returning()
			if (!updated) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Service status could not be updated",
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SERVICES,
					action: input.isActive ? "service.enable" : "service.disable",
					actorId,
					targetType: "service",
					targetId: String(current.id),
					oldValue: this.auditValue(current),
					newValue: this.auditValue(updated),
				},
				tx
			)
			return serviceResponse(updated, category)
		})
	}

	private auditValue(service: ServiceRow) {
		return {
			id: service.id,
			serviceCategoryId: service.serviceCategoryId,
			code: service.code,
			name: service.name,
			feeCentavos: service.feeCentavos,
			requiresApproval: service.requiresApproval,
			documentType: service.documentType,
			autoIssue: service.autoIssue,
			isActive: service.isActive,
		}
	}
}

import { Injectable } from "@nestjs/common"
import { ORPCError } from "@orpc/server"
import { and, asc, eq, gte, lt } from "drizzle-orm"

import type {
	GenerateSlotsInput,
	GenerateSlotsResponse,
	ListSlotsRequest,
	ScheduleSlot,
	SetSlotDisabledRequest,
	SlotAvailability,
	UpdateSlotCapacityRequest,
} from "@repo/contracts"
import { locations, scheduleSlots, services } from "@repo/db/schema"

import { AuditService } from "@/common/audit/audit.service"
import { db } from "@/common/database/database.client"

import {
	eachDate,
	manilaDateOf,
	manilaDateTimeToUtc,
	manilaMonthRange,
	weekdayOf,
} from "./manila-time"

const AUDIT_DOMAIN_SCHEDULES = "schedules"

/** The most slots one generate request may create. */
export const MAX_SLOTS_PER_REQUEST = 2000

/** Rows per INSERT, to stay well under PostgreSQL's parameter limit. */
const INSERT_CHUNK_SIZE = 500

type SlotRow = typeof scheduleSlots.$inferSelect

/** The start and end of every slot a generate request asks for, in UTC. */
export function slotTimesFor(
	input: Pick<GenerateSlotsInput, "startDate" | "endDate" | "weekdays" | "windows">
): { startsAt: Date; endsAt: Date }[] {
	const weekdays = new Set(input.weekdays)
	const times: { startsAt: Date; endsAt: Date }[] = []
	for (const date of eachDate(input.startDate, input.endDate)) {
		if (!weekdays.has(weekdayOf(date))) continue
		for (const window of input.windows) {
			times.push({
				startsAt: manilaDateTimeToUtc(date, window.startTime),
				endsAt: manilaDateTimeToUtc(date, window.endTime),
			})
		}
	}
	return times
}

/**
 * What a client sees for one slot. A slot cannot be booked when it is full,
 * disabled, or has already started.
 */
export function slotAvailability(slot: SlotRow, now: Date): SlotAvailability {
	const remainingCapacity = Math.max(0, slot.capacity - slot.bookedCount)
	return {
		id: slot.id,
		startsAt: slot.startsAt,
		endsAt: slot.endsAt,
		remainingCapacity,
		isAvailable:
			!slot.isDisabled && remainingCapacity > 0 && slot.startsAt.getTime() > now.getTime(),
	}
}

function bookingCount(count: number): string {
	return count === 1 ? "1 booking" : `${count} bookings`
}

/**
 * Schedule Slots (SCH-1): capacity for one Service at one Location on one date
 * and time window.
 *
 * This service creates slots and edits `capacity` and `is_disabled`. It never
 * writes `booked_count`: only `BookingService` changes it, with a row lock
 * (DOMAIN-MODEL.md section 6). Every change is audited in the same transaction.
 */
@Injectable()
export class SchedulesService {
	constructor(private readonly auditService: AuditService) {}

	/** Every slot for a service at a location in one Manila month, disabled ones included. */
	async listSlots(input: ListSlotsRequest): Promise<ScheduleSlot[]> {
		const { start, end } = manilaMonthRange(input.month)
		return db
			.select()
			.from(scheduleSlots)
			.where(
				and(
					eq(scheduleSlots.serviceId, input.serviceId),
					eq(scheduleSlots.locationId, input.locationId),
					gte(scheduleSlots.startsAt, start),
					lt(scheduleSlots.startsAt, end)
				)
			)
			.orderBy(asc(scheduleSlots.startsAt))
	}

	/**
	 * Remaining capacity per slot for a signed-in client choosing a date.
	 * A disabled service or location has nothing to book, so it returns no slots.
	 */
	async listAvailability(
		input: ListSlotsRequest,
		now: Date = new Date()
	): Promise<SlotAvailability[]> {
		const { start, end } = manilaMonthRange(input.month)
		const rows = await db
			.select({ slot: scheduleSlots })
			.from(scheduleSlots)
			.innerJoin(services, eq(scheduleSlots.serviceId, services.id))
			.innerJoin(locations, eq(scheduleSlots.locationId, locations.id))
			.where(
				and(
					eq(scheduleSlots.serviceId, input.serviceId),
					eq(scheduleSlots.locationId, input.locationId),
					eq(services.isActive, true),
					eq(locations.isActive, true),
					gte(scheduleSlots.startsAt, start),
					lt(scheduleSlots.startsAt, end)
				)
			)
			.orderBy(asc(scheduleSlots.startsAt))
		return rows.map(({ slot }) => slotAvailability(slot, now))
	}

	/**
	 * Create one slot per date and time window. A slot that already exists for
	 * the same service, location and start time is skipped, so the same request
	 * can be sent twice without creating duplicates.
	 */
	async generateSlots(
		input: GenerateSlotsInput,
		actorId: string | null,
		now: Date = new Date()
	): Promise<GenerateSlotsResponse> {
		if (input.startDate < manilaDateOf(now)) {
			throw new ORPCError("BAD_REQUEST", { message: "Start date cannot be in the past" })
		}

		const times = slotTimesFor(input)
		if (times.length === 0) {
			throw new ORPCError("BAD_REQUEST", {
				message: "No date in that range falls on the chosen days of the week",
			})
		}
		if (times.length > MAX_SLOTS_PER_REQUEST) {
			throw new ORPCError("BAD_REQUEST", {
				message: `That would create ${times.length} slots. Create ${MAX_SLOTS_PER_REQUEST} or fewer at a time by shortening the date range.`,
			})
		}

		return db.transaction(async tx => {
			const [service] = await tx
				.select({ id: services.id, name: services.name, isActive: services.isActive })
				.from(services)
				.where(eq(services.id, input.serviceId))
			if (!service) {
				throw new ORPCError("NOT_FOUND", { message: `Service "${input.serviceId}" was not found` })
			}
			if (!service.isActive) {
				throw new ORPCError("CONFLICT", {
					message: `Service "${service.name}" is disabled. Enable it before adding slots.`,
				})
			}

			const [location] = await tx
				.select({ id: locations.id, name: locations.name, isActive: locations.isActive })
				.from(locations)
				.where(eq(locations.id, input.locationId))
			if (!location) {
				throw new ORPCError("NOT_FOUND", {
					message: `Location "${input.locationId}" was not found`,
				})
			}
			if (!location.isActive) {
				throw new ORPCError("CONFLICT", {
					message: `Location "${location.name}" is disabled. Enable it before adding slots.`,
				})
			}

			let createdCount = 0
			for (let index = 0; index < times.length; index += INSERT_CHUNK_SIZE) {
				const created = await tx
					.insert(scheduleSlots)
					.values(
						times.slice(index, index + INSERT_CHUNK_SIZE).map(({ startsAt, endsAt }) => ({
							serviceId: input.serviceId,
							locationId: input.locationId,
							startsAt,
							endsAt,
							capacity: input.capacity,
						}))
					)
					.onConflictDoNothing({
						target: [scheduleSlots.serviceId, scheduleSlots.locationId, scheduleSlots.startsAt],
					})
					.returning({ id: scheduleSlots.id })
				createdCount += created.length
			}
			const skippedCount = times.length - createdCount

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SCHEDULES,
					action: "schedule.slots.generate",
					actorId,
					targetType: "schedule_slot",
					targetId: null,
					oldValue: null,
					newValue: {
						serviceId: input.serviceId,
						locationId: input.locationId,
						startDate: input.startDate,
						endDate: input.endDate,
						weekdays: input.weekdays,
						windows: input.windows,
						capacity: input.capacity,
						createdCount,
						skippedCount,
					},
				},
				tx
			)
			return { createdCount, skippedCount }
		})
	}

	/** Change capacity. It may not go below the bookings already made. */
	async updateSlotCapacity(
		input: UpdateSlotCapacityRequest,
		actorId: string | null
	): Promise<ScheduleSlot> {
		return db.transaction(async tx => {
			// Locked so a booking cannot take a place between this check and the update.
			const [current] = await tx
				.select()
				.from(scheduleSlots)
				.where(eq(scheduleSlots.id, input.id))
				.for("update")
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Schedule slot "${input.id}" was not found` })
			}
			if (input.capacity < current.bookedCount) {
				throw new ORPCError("CONFLICT", {
					message: `Capacity cannot be lower than the ${bookingCount(current.bookedCount)} already made for this slot.`,
				})
			}

			const [updated] = await tx
				.update(scheduleSlots)
				.set({ capacity: input.capacity, updatedAt: new Date() })
				.where(eq(scheduleSlots.id, current.id))
				.returning()
			if (!updated) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Schedule slot capacity could not be updated",
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SCHEDULES,
					action: "schedule.slot.capacity.update",
					actorId,
					targetType: "schedule_slot",
					targetId: String(current.id),
					oldValue: this.auditValue(current),
					newValue: this.auditValue(updated),
				},
				tx
			)
			return updated
		})
	}

	/** Disable or re-enable a slot. Existing bookings are not touched. */
	async setSlotDisabled(
		input: SetSlotDisabledRequest,
		actorId: string | null
	): Promise<ScheduleSlot> {
		return db.transaction(async tx => {
			const [current] = await tx.select().from(scheduleSlots).where(eq(scheduleSlots.id, input.id))
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Schedule slot "${input.id}" was not found` })
			}

			const [updated] = await tx
				.update(scheduleSlots)
				.set({ isDisabled: input.isDisabled, updatedAt: new Date() })
				.where(eq(scheduleSlots.id, current.id))
				.returning()
			if (!updated) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Schedule slot status could not be updated",
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SCHEDULES,
					action: input.isDisabled ? "schedule.slot.disable" : "schedule.slot.enable",
					actorId,
					targetType: "schedule_slot",
					targetId: String(current.id),
					oldValue: this.auditValue(current),
					newValue: this.auditValue(updated),
				},
				tx
			)
			return updated
		})
	}

	/** Delete a slot that nobody has booked. A booked slot is disabled instead. */
	async deleteSlot(id: number, actorId: string | null): Promise<{ success: true; id: number }> {
		return db.transaction(async tx => {
			// Locked so a booking cannot land on the slot while it is being deleted.
			const [current] = await tx
				.select()
				.from(scheduleSlots)
				.where(eq(scheduleSlots.id, id))
				.for("update")
			if (!current) {
				throw new ORPCError("NOT_FOUND", { message: `Schedule slot "${id}" was not found` })
			}
			if (current.bookedCount > 0) {
				throw new ORPCError("CONFLICT", {
					message: `This slot has ${bookingCount(current.bookedCount)} and cannot be deleted. Disable it instead.`,
				})
			}

			await tx.delete(scheduleSlots).where(eq(scheduleSlots.id, current.id))

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_SCHEDULES,
					action: "schedule.slot.delete",
					actorId,
					targetType: "schedule_slot",
					targetId: String(current.id),
					oldValue: this.auditValue(current),
					newValue: null,
				},
				tx
			)
			return { success: true as const, id: current.id }
		})
	}

	private auditValue(slot: SlotRow) {
		return {
			id: slot.id,
			serviceId: slot.serviceId,
			locationId: slot.locationId,
			startsAt: slot.startsAt.toISOString(),
			endsAt: slot.endsAt.toISOString(),
			capacity: slot.capacity,
			bookedCount: slot.bookedCount,
			isDisabled: slot.isDisabled,
		}
	}
}

import { Injectable } from "@nestjs/common"
import { ORPCError } from "@orpc/server"
import { and, asc, eq, sql } from "drizzle-orm"

import type { Booking, BookingItemStatus, BookingStatus, CreateBookingInput } from "@repo/contracts"
import { bookingItems, bookings, locations, scheduleSlots, services, users } from "@repo/db/schema"

import { AuditService } from "@/common/audit/audit.service"
import { db } from "@/common/database/database.client"
import { env } from "@/config/env.config"

import { canTransitionItem, isTerminalItem } from "./booking-item-state"
import { canTransition, initialStatus } from "./booking-state"
import { generateReferenceNumber } from "./reference-number"

const AUDIT_DOMAIN_BOOKINGS = "bookings"

/** How many times to draw a new Reference Number when one is already taken. */
const MAX_REFERENCE_ATTEMPTS = 5

const MS_PER_MINUTE = 60_000

export type Transaction = Parameters<Parameters<typeof db.transaction>[0]>[0]

/** What happened when staff tried to check a Booking in (PRC-1). */
export type CheckInResult =
	| { ok: true; bookingItemIds: number[]; checkedInIds: number[] }
	| { ok: false; reason: "payment_required"; bookingStatus: BookingStatus }
	| { ok: false; reason: "no_items_here" }

/**
 * Bookings (BKG-1). The only code that changes `bookings.status`,
 * `booking_items.status` (PRC-1, CRT-1) and `schedule_slots.booked_count`
 * (DOMAIN-MODEL.md section 6).
 */
@Injectable()
export class BookingService {
	constructor(private readonly auditService: AuditService) {}

	/**
	 * Book one Service in one Schedule Slot (Flow B, steps 1 to 3).
	 *
	 * The slot row is locked before its capacity is read, so when two clients
	 * confirm the last place at the same moment the second waits, then sees the
	 * slot full. The booking, its item, the count and the audit entry commit
	 * together or not at all.
	 */
	async create(
		input: CreateBookingInput,
		actorId: string | null,
		now: Date = new Date()
	): Promise<Booking> {
		if (!actorId) {
			throw new ORPCError("UNAUTHORIZED", { message: "Sign in to book" })
		}

		return db.transaction(async tx => {
			// Locks the client's own row so two requests from the same client are
			// handled one after the other, and the unpaid-booking check below holds.
			await tx.select({ id: users.id }).from(users).where(eq(users.id, actorId)).for("update")

			const [unpaid] = await tx
				.select({ referenceNumber: bookings.referenceNumber })
				.from(bookings)
				.where(and(eq(bookings.clientId, actorId), eq(bookings.status, "pending_payment")))
				.limit(1)
			if (unpaid) {
				throw new ORPCError("CONFLICT", {
					message: `You already have a booking awaiting payment (${unpaid.referenceNumber}). Pay or cancel it before booking again.`,
				})
			}

			const [slot] = await tx
				.select()
				.from(scheduleSlots)
				.where(eq(scheduleSlots.id, input.scheduleSlotId))
				.for("update")
			if (!slot) {
				throw new ORPCError("NOT_FOUND", {
					message: `Schedule slot "${input.scheduleSlotId}" was not found`,
				})
			}
			if (slot.isDisabled) {
				throw new ORPCError("CONFLICT", { message: "This slot is no longer available" })
			}
			if (slot.startsAt.getTime() <= now.getTime()) {
				throw new ORPCError("CONFLICT", { message: "This slot has already started" })
			}
			if (slot.bookedCount >= slot.capacity) {
				throw new ORPCError("CONFLICT", { message: "This slot is full" })
			}

			const [service] = await tx.select().from(services).where(eq(services.id, slot.serviceId))
			if (!service?.isActive) {
				throw new ORPCError("CONFLICT", { message: "This service is not available for booking" })
			}
			const [location] = await tx.select().from(locations).where(eq(locations.id, slot.locationId))
			if (!location?.isActive) {
				throw new ORPCError("CONFLICT", { message: "This location is not available for booking" })
			}

			const referenceNumber = await this.unusedReferenceNumber(tx, now)
			const holdExpiresAt = new Date(now.getTime() + env.BOOKING_HOLD_MINUTES * MS_PER_MINUTE)

			const [booking] = await tx
				.insert(bookings)
				.values({
					referenceNumber,
					clientId: actorId,
					status: initialStatus("regular"),
					kind: "regular",
					totalCentavos: service.feeCentavos,
					holdExpiresAt,
					createdBy: actorId,
				})
				.returning()
			if (!booking) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", { message: "Booking could not be created" })
			}

			const [item] = await tx
				.insert(bookingItems)
				.values({
					bookingId: booking.id,
					serviceId: slot.serviceId,
					scheduleSlotId: slot.id,
					locationId: slot.locationId,
					feeCentavos: service.feeCentavos,
					status: "booked",
				})
				.returning()
			if (!item) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", { message: "Booking could not be created" })
			}

			// Counted in the database, on the locked row, so it cannot be lost.
			await tx
				.update(scheduleSlots)
				.set({ bookedCount: sql`${scheduleSlots.bookedCount} + 1`, updatedAt: now })
				.where(eq(scheduleSlots.id, slot.id))

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_BOOKINGS,
					action: "booking.create",
					actorId,
					targetType: "booking",
					targetId: String(booking.id),
					oldValue: null,
					newValue: {
						id: booking.id,
						referenceNumber: booking.referenceNumber,
						clientId: booking.clientId,
						status: booking.status,
						kind: booking.kind,
						totalCentavos: booking.totalCentavos,
						holdExpiresAt: holdExpiresAt.toISOString(),
						items: [
							{
								id: item.id,
								serviceId: item.serviceId,
								scheduleSlotId: item.scheduleSlotId,
								locationId: item.locationId,
								feeCentavos: item.feeCentavos,
								status: item.status,
							},
						],
					},
				},
				tx
			)

			return {
				id: booking.id,
				referenceNumber: booking.referenceNumber,
				status: booking.status,
				kind: booking.kind,
				totalCentavos: booking.totalCentavos,
				holdExpiresAt: booking.holdExpiresAt,
				createdAt: booking.createdAt,
				items: [
					{
						id: item.id,
						serviceId: item.serviceId,
						serviceName: service.name,
						locationId: item.locationId,
						locationName: location.name,
						scheduleSlotId: item.scheduleSlotId,
						startsAt: slot.startsAt,
						endsAt: slot.endsAt,
						feeCentavos: item.feeCentavos,
						status: item.status,
					},
				],
			}
		})
	}

	/**
	 * Mark a Booking paid (Flow C step 5, PAY-1).
	 *
	 * Called only by `PaymentService.confirm`, inside its transaction, so the
	 * Payment and the Booking change together or not at all. Locks the Booking
	 * row first. Returns false and changes nothing when the Booking can no
	 * longer be paid (for example it was cancelled while the client was at
	 * checkout); the caller then flags the Payment for review.
	 */
	async markPaid(
		tx: Transaction,
		bookingId: number,
		actorId: string | null,
		now: Date = new Date()
	): Promise<boolean> {
		const [booking] = await tx
			.select({ id: bookings.id, status: bookings.status })
			.from(bookings)
			.where(eq(bookings.id, bookingId))
			.for("update")
		if (!booking) {
			throw new ORPCError("NOT_FOUND", { message: `Booking "${bookingId}" was not found` })
		}
		if (!canTransition(booking.status, "paid")) {
			return false
		}

		// A paid booking keeps its slot, so the hold no longer applies.
		await tx
			.update(bookings)
			.set({ status: "paid", holdExpiresAt: null, updatedAt: now })
			.where(eq(bookings.id, bookingId))

		await this.auditService.record(
			{
				domain: AUDIT_DOMAIN_BOOKINGS,
				action: "booking.paid",
				actorId,
				targetType: "booking",
				targetId: String(bookingId),
				oldValue: { status: booking.status },
				newValue: { status: "paid" },
			},
			tx
		)
		return true
	}

	/**
	 * Check a client in at one Location (PRC-1, state machine 5.2).
	 *
	 * Locks the Booking, then its items at that Location. Annex B: no service
	 * is rendered unless payment is completed, so nothing changes unless the
	 * Booking is `paid`. Items already past `booked` are left as they are, so
	 * scanning the same QR twice is harmless. Runs in the caller's transaction.
	 */
	async checkInItems(
		tx: Transaction,
		bookingId: number,
		locationId: number,
		actorId: string,
		now: Date = new Date()
	): Promise<CheckInResult> {
		const [booking] = await tx
			.select({ id: bookings.id, status: bookings.status })
			.from(bookings)
			.where(eq(bookings.id, bookingId))
			.for("update")
		if (!booking) {
			throw new ORPCError("NOT_FOUND", { message: `Booking "${bookingId}" was not found` })
		}

		const items = await tx
			.select({ id: bookingItems.id, status: bookingItems.status })
			.from(bookingItems)
			.where(and(eq(bookingItems.bookingId, bookingId), eq(bookingItems.locationId, locationId)))
			.orderBy(asc(bookingItems.id))
			.for("update")
		if (items.length === 0) {
			return { ok: false, reason: "no_items_here" }
		}
		if (booking.status !== "paid") {
			return { ok: false, reason: "payment_required", bookingStatus: booking.status }
		}

		const toCheckIn = items.filter(item => canTransitionItem(item.status, "checked_in"))
		for (const item of toCheckIn) {
			await tx
				.update(bookingItems)
				.set({ status: "checked_in", updatedAt: now })
				.where(eq(bookingItems.id, item.id))
			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_BOOKINGS,
					action: "booking_item.check_in",
					actorId,
					targetType: "booking_item",
					targetId: String(item.id),
					oldValue: { status: item.status },
					newValue: { status: "checked_in" },
				},
				tx
			)
		}

		return {
			ok: true,
			bookingItemIds: items.map(item => item.id),
			checkedInIds: toCheckIn.map(item => item.id),
		}
	}

	/**
	 * Move one Booking Item along state machine 5.2 (PRC-1).
	 *
	 * Locks the item, refuses any move that is not an arrow in the machine or
	 * that starts from a status the caller does not expect, then writes the
	 * item and its Audit Entry in the caller's transaction. Asking for
	 * `cleared` from `in_assessment` gives `pending_head_approval` instead when
	 * the Service has `requires_approval`. Returns the status actually set.
	 */
	async transitionItem(
		tx: Transaction,
		bookingItemId: number,
		to: BookingItemStatus,
		options: { actorId: string; allowedFrom: readonly BookingItemStatus[]; now?: Date }
	): Promise<{ from: BookingItemStatus; to: BookingItemStatus }> {
		const now = options.now ?? new Date()
		const [item] = await tx
			.select({
				id: bookingItems.id,
				status: bookingItems.status,
				serviceId: bookingItems.serviceId,
			})
			.from(bookingItems)
			.where(eq(bookingItems.id, bookingItemId))
			.for("update")
		if (!item) {
			throw new ORPCError("NOT_FOUND", { message: `Booking item "${bookingItemId}" was not found` })
		}

		let target = to
		if (to === "cleared" && item.status === "in_assessment") {
			const [service] = await tx
				.select({ requiresApproval: services.requiresApproval })
				.from(services)
				.where(eq(services.id, item.serviceId))
			if (service?.requiresApproval) target = "pending_head_approval"
		}

		if (!options.allowedFrom.includes(item.status) || !canTransitionItem(item.status, target)) {
			throw new ORPCError("CONFLICT", {
				message: `This item is ${item.status.replaceAll("_", " ")} and cannot be moved to ${target.replaceAll("_", " ")}`,
			})
		}

		await tx
			.update(bookingItems)
			.set({ status: target, updatedAt: now })
			.where(eq(bookingItems.id, item.id))

		await this.auditService.record(
			{
				domain: AUDIT_DOMAIN_BOOKINGS,
				action: `booking_item.${target}`,
				actorId: options.actorId,
				targetType: "booking_item",
				targetId: String(item.id),
				oldValue: { status: item.status },
				newValue: { status: target },
			},
			tx
		)

		return { from: item.status, to: target }
	}

	/**
	 * Complete a paid Booking once every item is final (state machine 5.1:
	 * `paid` to `completed` when each item is `issued`, `rejected` or
	 * `no_show`). Called by `DocumentService.issue` in its transaction (CRT-1).
	 * Returns false and changes nothing while any item is still open, or when
	 * the Booking is not `paid`.
	 */
	async completeIfAllItemsFinal(
		tx: Transaction,
		bookingId: number,
		actorId: string | null,
		now: Date = new Date()
	): Promise<boolean> {
		const [booking] = await tx
			.select({ id: bookings.id, status: bookings.status })
			.from(bookings)
			.where(eq(bookings.id, bookingId))
			.for("update")
		if (!booking) {
			throw new ORPCError("NOT_FOUND", { message: `Booking "${bookingId}" was not found` })
		}

		// Locked too, so no item changes between this check and the update.
		const items = await tx
			.select({ status: bookingItems.status })
			.from(bookingItems)
			.where(eq(bookingItems.bookingId, bookingId))
			.orderBy(asc(bookingItems.id))
			.for("update")
		if (items.length === 0 || !items.every(item => isTerminalItem(item.status))) return false
		if (!canTransition(booking.status, "completed")) return false

		await tx
			.update(bookings)
			.set({ status: "completed", updatedAt: now })
			.where(eq(bookings.id, bookingId))
		await this.auditService.record(
			{
				domain: AUDIT_DOMAIN_BOOKINGS,
				action: "booking.completed",
				actorId,
				targetType: "booking",
				targetId: String(bookingId),
				oldValue: { status: booking.status },
				newValue: { status: "completed" },
			},
			tx
		)
		return true
	}

	/** Draw Reference Numbers until one is not already in use. */
	private async unusedReferenceNumber(tx: Transaction, now: Date): Promise<string> {
		for (let attempt = 0; attempt < MAX_REFERENCE_ATTEMPTS; attempt++) {
			const referenceNumber = generateReferenceNumber(now)
			const [taken] = await tx
				.select({ id: bookings.id })
				.from(bookings)
				.where(eq(bookings.referenceNumber, referenceNumber))
				.limit(1)
			if (!taken) return referenceNumber
		}
		throw new ORPCError("INTERNAL_SERVER_ERROR", {
			message: "A reference number could not be generated. Please try again.",
		})
	}
}

import { Controller, UseInterceptors } from "@nestjs/common"

}

			this.bookingService.create(input, actorId(session))
		)
	}
}

===== apps/backend/src/modules/v1/processing/processing.service.ts =====
import { Injectable } from "@nestjs/common"
import { ORPCError } from "@orpc/server"
import { and, asc, desc, eq, gte, inArray, lt } from "drizzle-orm"

import type {
	BookingItemIdParam,
	BookingStatus,
	CheckInInput,
	CheckInResponse,
	ProcessingItem,
	ProcessingQueue,
	ProcessingQueueRequest,
	QueuePaymentStatus,
	RecordAssessmentInput,
} from "@repo/contracts"
import {
	assessments,
	bookingItems,
	bookings,
	documents,
	locations,
	scheduleSlots,
	services,
	users,
	userScopes,
} from "@repo/db/schema"

import { AuditService } from "@/common/audit/audit.service"
import { db } from "@/common/database/database.client"

import { BookingService } from "../bookings/bookings.service"
import { DocumentService } from "../certificates/documents.service"
import { manilaDateOf, manilaDateTimeToUtc } from "../schedules/manila-time"

const AUDIT_DOMAIN_PROCESSING = "processing"

const MS_PER_DAY = 86_400_000

/**
 * Bookings staff work on. A cancelled, rejected or refunded Booking has
 * nothing left to assess, so it stays out of the queue.
 */
const QUEUE_BOOKING_STATUSES: BookingStatus[] = ["pending_payment", "paid", "completed"]

/** Annex B: Paid is `paid` or `completed`. */
function paymentStatusOf(status: BookingStatus): QueuePaymentStatus {
	return status === "paid" || status === "completed" ? "paid" : "unpaid"
}

/** The same answer for "not there" and "not yours", so other Locations cannot be probed. */
function itemNotFound() {
	return new ORPCError("NOT_FOUND", { message: "Booking item not found at your location" })
}

interface StaffLocation {
	id: number
	name: string
}

/**
 * BOQ staff assessment (PRC-1, TOR §VIII.3).
 *
 * Every read and write is limited to the staff user's Location, read from
 * `user_scopes` (DOMAIN-MODEL.md section 4) until ADM-2 adds `ScopeService`.
 * Item status changes go through `BookingService`; this service writes only
 * `assessments`.
 */
@Injectable()
export class ProcessingService {
	constructor(
		private readonly bookingService: BookingService,
		private readonly auditService: AuditService,
		private readonly documentService: DocumentService
	) {}

	/** Booking Items at the caller's Location for one Manila date, earliest first. */
	async queue(
		input: ProcessingQueueRequest,
		actorId: string | null,
		now: Date = new Date()
	): Promise<ProcessingQueue> {
		const staffId = requireActor(actorId)
		const date = input.date ?? manilaDateOf(now)
		const location = await this.staffLocation(staffId)
		if (!location) {
			return { location: null, date, items: [] }
		}

		const dayStart = manilaDateTimeToUtc(date, "00:00")
		const dayEnd = new Date(dayStart.getTime() + MS_PER_DAY)

		const rows = await db
			.select({
				bookingItemId: bookingItems.id,
				status: bookingItems.status,
				referenceNumber: bookings.referenceNumber,
				bookingStatus: bookings.status,
				clientName: users.name,
				serviceName: services.name,
				startsAt: scheduleSlots.startsAt,
				endsAt: scheduleSlots.endsAt,
			})
			.from(bookingItems)
			.innerJoin(bookings, eq(bookings.id, bookingItems.bookingId))
			.innerJoin(users, eq(users.id, bookings.clientId))
			.innerJoin(services, eq(services.id, bookingItems.serviceId))
			.innerJoin(scheduleSlots, eq(scheduleSlots.id, bookingItems.scheduleSlotId))
			.where(
				and(
					eq(bookingItems.locationId, location.id),
					gte(scheduleSlots.startsAt, dayStart),
					lt(scheduleSlots.startsAt, dayEnd),
					inArray(bookings.status, QUEUE_BOOKING_STATUSES)
				)
			)
			.orderBy(asc(scheduleSlots.startsAt), asc(bookings.referenceNumber))

		return {
			location,
			date,
			items: rows.map(row => ({
				...row,
				paymentStatus: paymentStatusOf(row.bookingStatus),
			})),
		}
	}

	/** One Booking Item with the client's profile and assessment history. */
	async item(input: BookingItemIdParam, actorId: string | null): Promise<ProcessingItem> {
		const staffId = requireActor(actorId)
		await this.requireItemInScope(input.bookingItemId, staffId, "booking_item.read")
		return this.loadItem(input.bookingItemId)
	}

	/**
	 * Check a client in by Reference Number (from the QR code or typed).
	 * Refused with "Payment required" unless the Booking is paid (Annex B).
	 */
	async checkIn(input: CheckInInput, actorId: string | null): Promise<CheckInResponse> {
		const staffId = requireActor(actorId)
		const location = await this.staffLocation(staffId)

		const [booking] = await db
			.select({ id: bookings.id })
			.from(bookings)
			.where(eq(bookings.referenceNumber, input.referenceNumber))
			.limit(1)
		if (!booking) {
			throw new ORPCError("NOT_FOUND", {
				message: `No booking has the reference number ${input.referenceNumber}`,
			})
		}
		if (!location) {
			await this.recordDenial(
				staffId,
				"booking_item.check_in",
				"booking",
				booking.id,
				"no_location"
			)
			throw new ORPCError("NOT_FOUND", {
				message: `No booking ${input.referenceNumber} at your location`,
			})
		}

		const result = await db.transaction(tx =>
			this.bookingService.checkInItems(tx, booking.id, location.id, staffId)
		)

		if (!result.ok && result.reason === "no_items_here") {
			await this.recordDenial(
				staffId,
				"booking_item.check_in",
				"booking",
				booking.id,
				"outside_scope"
			)
			throw new ORPCError("NOT_FOUND", {
				message: `No booking ${input.referenceNumber} at your location`,
			})
		}
		if (!result.ok) {
			await this.recordDenial(
				staffId,
				"booking_item.check_in",
				"booking",
				booking.id,
				"payment_required"
			)
			throw new ORPCError("CONFLICT", {
				message: `Payment required: booking ${input.referenceNumber} is not paid yet, so the client cannot be checked in.`,
			})
		}

		const items = await Promise.all(result.bookingItemIds.map(id => this.loadItem(id)))
		return { items }
	}

	/** Start assessing: `checked_in` (or `returned`, after the client corrects it) to `in_assessment`. */
	async open(input: BookingItemIdParam, actorId: string | null): Promise<ProcessingItem> {
		const staffId = requireActor(actorId)
		await this.requireItemInScope(input.bookingItemId, staffId, "booking_item.open")

		await db.transaction(tx =>
			this.bookingService.transitionItem(tx, input.bookingItemId, "in_assessment", {
				actorId: staffId,
				allowedFrom: ["checked_in", "returned"],
			})
		)
		return this.loadItem(input.bookingItemId)
	}

	/**
	 * Record an Assessment and move the item: cleared (or pending head
	 * approval), returned for correction, or rejected. The Assessment row, the
	 * item's new status and both Audit Entries commit together.
	 *
	 * CRT-1: when the item is cleared and its Service issues a document
	 * automatically, the document is issued in this same transaction, recorded
	 * against this officer. If the certificate details are missing, nothing is
	 * saved.
	 */
	async assess(input: RecordAssessmentInput, actorId: string | null): Promise<ProcessingItem> {
		const staffId = requireActor(actorId)
		await this.requireItemInScope(input.bookingItemId, staffId, "assessment.create")

		await db.transaction(async tx => {
			const moved = await this.bookingService.transitionItem(
				tx,
				input.bookingItemId,
				input.outcome,
				{ actorId: staffId, allowedFrom: ["in_assessment"] }
			)

			if (moved.to === "cleared") {
				await this.documentService.issueOnClearance(tx, {
					bookingItemId: input.bookingItemId,
					actorId: staffId,
					details: input.document ?? null,
				})
			}

			const notes = input.notes.length > 0 ? input.notes : null
			const [saved] = await tx
				.insert(assessments)
				.values({
					bookingItemId: input.bookingItemId,
					staffId,
					outcome: input.outcome,
					notes,
					corrections: input.corrections,
					// Set here, in UTC, rather than by the database clock (DOMAIN-MODEL.md: timestamps are UTC).
					createdAt: new Date(),
				})
				.returning({ id: assessments.id })
			if (!saved) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "The assessment could not be saved",
				})
			}

			await this.auditService.record(
				{
					domain: AUDIT_DOMAIN_PROCESSING,
					action: "assessment.create",
					actorId: staffId,
					targetType: "booking_item",
					targetId: String(input.bookingItemId),
					oldValue: { status: moved.from },
					newValue: {
						assessmentId: saved.id,
						outcome: input.outcome,
						status: moved.to,
						notes,
						corrections: input.corrections,
					},
				},
				tx
			)
		})

		return this.loadItem(input.bookingItemId)
	}

	/**
	 * The staff user's Location from `user_scopes`, or null when they have none.
	 *
	 * PRC-1 supports one Location per staff user: the first scope row that names
	 * a Location. A row with a null `location_id` ("all Locations") and several
	 * Locations per user are handled by ADM-2's `ScopeService`.
	 */
	private async staffLocation(staffId: string): Promise<StaffLocation | null> {
		const [row] = await db
			.select({ id: locations.id, name: locations.name })
			.from(userScopes)
			.innerJoin(locations, eq(locations.id, userScopes.locationId))
			.where(eq(userScopes.userId, staffId))
			.orderBy(asc(userScopes.id))
			.limit(1)
		return row ?? null
	}

	/**
	 * 404 unless the item exists and is at the caller's Location. A refused
	 * attempt on an item that does exist is audited as denied.
	 */
	private async requireItemInScope(
		bookingItemId: number,
		staffId: string,
		action: string
	): Promise<void> {
		const [item] = await db
			.select({ id: bookingItems.id, locationId: bookingItems.locationId })
			.from(bookingItems)
			.where(eq(bookingItems.id, bookingItemId))
			.limit(1)
		if (!item) throw itemNotFound()

		const location = await this.staffLocation(staffId)
		if (!location || location.id !== item.locationId) {
			await this.recordDenial(
				staffId,
				action,
				"booking_item",
				bookingItemId,
				location ? "outside_scope" : "no_location"
			)
			throw itemNotFound()
		}
	}

	/** Everything the assessment screen shows. Call only after the scope check. */
	private async loadItem(bookingItemId: number): Promise<ProcessingItem> {
		const [row] = await db
			.select({
				bookingItemId: bookingItems.id,
				status: bookingItems.status,
				feeCentavos: bookingItems.feeCentavos,
				bookingId: bookings.id,
				referenceNumber: bookings.referenceNumber,
				bookingStatus: bookings.status,
				kind: bookings.kind,
				totalCentavos: bookings.totalCentavos,
				serviceId: services.id,
				serviceName: services.name,
				requiresApproval: services.requiresApproval,
				documentType: services.documentType,
				autoIssue: services.autoIssue,
				documentId: documents.id,
				documentNumber: documents.documentNumber,
				locationId: locations.id,
				locationName: locations.name,
				startsAt: scheduleSlots.startsAt,
				endsAt: scheduleSlots.endsAt,
				clientId: users.id,
				clientName: users.name,
				clientEmail: users.email,
				clientMobile: users.mobile,
				clientBirthDate: users.birthDate,
				clientGender: users.gender,
				clientNationality: users.nationality,
			})
			.from(bookingItems)
			.innerJoin(bookings, eq(bookings.id, bookingItems.bookingId))
			.innerJoin(users, eq(users.id, bookings.clientId))
			.innerJoin(services, eq(services.id, bookingItems.serviceId))
			.innerJoin(locations, eq(locations.id, bookingItems.locationId))
			.innerJoin(scheduleSlots, eq(scheduleSlots.id, bookingItems.scheduleSlotId))
			.leftJoin(
				documents,
				and(eq(documents.bookingItemId, bookingItems.id), eq(documents.status, "valid"))
			)
			.where(eq(bookingItems.id, bookingItemId))
			.limit(1)
		if (!row) throw itemNotFound()

		const history = await db
			.select({
				id: assessments.id,
				outcome: assessments.outcome,
				notes: assessments.notes,
				corrections: assessments.corrections,
				staffId: assessments.staffId,
				staffName: users.name,
				createdAt: assessments.createdAt,
			})
			.from(assessments)
			.innerJoin(users, eq(users.id, assessments.staffId))
			.where(eq(assessments.bookingItemId, bookingItemId))
			.orderBy(desc(assessments.createdAt), desc(assessments.id))

		return {
			bookingItemId: row.bookingItemId,
			status: row.status,
			booking: {
				id: row.bookingId,
				referenceNumber: row.referenceNumber,
				status: row.bookingStatus,
				kind: row.kind,
				totalCentavos: row.totalCentavos,
			},
			paymentStatus: paymentStatusOf(row.bookingStatus),
			service: {
				id: row.serviceId,
				name: row.serviceName,
				requiresApproval: row.requiresApproval,
				documentType: row.documentType,
				autoIssue: row.autoIssue,
			},
			document:
				row.documentId !== null && row.documentNumber !== null
					? { id: row.documentId, documentNumber: row.documentNumber }
					: null,
			location: { id: row.locationId, name: row.locationName },
			startsAt: row.startsAt,
			endsAt: row.endsAt,
			feeCentavos: row.feeCentavos,
			client: {
				id: row.clientId,
				name: row.clientName,
				email: row.clientEmail,
				mobile: row.clientMobile,
				birthDate: row.clientBirthDate,
				gender: row.clientGender,
				nationality: row.clientNationality,
			},
			// `form_submissions` arrives with the forms epic; until then there is nothing to show.
			formAnswers: {},
			assessments: history,
		}
	}

	/** Recorded outside any transaction, so the denial is kept when the request fails. */
	private async recordDenial(
		staffId: string,
		action: string,
		targetType: "booking" | "booking_item",
		targetId: number,
		reason: "outside_scope" | "no_location" | "payment_required"
	): Promise<void> {
		await this.auditService.record({
			domain: AUDIT_DOMAIN_PROCESSING,
			action,
			outcome: "denied",
			actorId: staffId,
			targetType,
			targetId: String(targetId),
			reason,
		})
	}
}

function requireActor(actorId: string | null): string {
	if (!actorId) {
		throw new ORPCError("UNAUTHORIZED", { message: "Sign in to use processing" })
	}
	return actorId
}

import { Controller, UseInterceptors } from "@nestjs/common"
import { Implement } from "@orpc/nest"
import { implement } from "@orpc/server"
import { Session, type UserSession } from "@thallesp/nestjs-better-auth"

import { v1 } from "@/config/api-versions.config"
import { RequirePermissions } from "@/shared/decorators/require-permissions.decorator"
import { IdempotencyInterceptor } from "@/shared/interceptors/idempotency.interceptor"

import { ProcessingService } from "./processing.service"

function actorId(session?: UserSession): string | null {
	return session?.user?.id ?? null
}

/**
 * BOQ staff assessment (PRC-1). RbacGuard checks the permission key (403 and
 * a denied Audit Entry without it); ProcessingService checks the Location.
 *
 * These routes use the normal throttle, not the strict one: staff make three
 * calls per client (check in, open, assess), and every route already needs a
 * staff session and a processing permission.
 */
@Controller()
export class ProcessingController {
	constructor(private readonly processingService: ProcessingService) {}

	@RequirePermissions("processing.queue:read")
	@Implement(v1.processing.queue)
	async queue(@Session() session?: UserSession) {
		return implement(v1.processing.queue).handler(({ input }) =>
			this.processingService.queue(input, actorId(session))
		)
	}

	@RequirePermissions("processing.queue:read")
	@Implement(v1.processing.item)
	async item(@Session() session?: UserSession) {
		return implement(v1.processing.item).handler(({ input }) =>
			this.processingService.item(input, actorId(session))
		)
	}

	// Scanning the same QR twice is harmless: items already checked in stay as they are.
	@RequirePermissions("processing.assessments:perform")
	@UseInterceptors(IdempotencyInterceptor)
	@Implement(v1.processing.checkIn)
	async checkIn(@Session() session?: UserSession) {
		return implement(v1.processing.checkIn).handler(({ input }) =>
			this.processingService.checkIn(input, actorId(session))
		)
	}

	@RequirePermissions("processing.assessments:perform")
	@UseInterceptors(IdempotencyInterceptor)
	@Implement(v1.processing.open)
	async open(@Session() session?: UserSession) {
		return implement(v1.processing.open).handler(({ input }) =>
			this.processingService.open(input, actorId(session))
		)
	}

	// Idempotent: a double click with the same Idempotency-Key replays the
	// first result instead of adding a second Assessment.
	@RequirePermissions("processing.assessments:perform")
	@UseInterceptors(IdempotencyInterceptor)
	@Implement(v1.processing.assess)
	async assess(@Session() session?: UserSession) {
		return implement(v1.processing.assess).handler(({ input }) =>
			this.processingService.assess(input, actorId(session))
		)
	}
}
